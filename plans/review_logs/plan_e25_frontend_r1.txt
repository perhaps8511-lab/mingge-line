## 2. `log.html`(卦記詳情頁)新增:金印按鈕

採用 **post-render DOM API 掛載**,完全不碰 `renderDetail()` 函式本體(不新增字串到它的 `html +=` 模板、不改它的回傳值),在 `renderDetail()` **執行完畢之後**,用原生 DOM API 找到已經渲染出來的 `.gua-header` 節點,`appendChild` 一個新建立的區塊。此法確保:
1. `renderDetail()` 原始碼**逐字元不變**(diff 對這個函式應為零差異)。
2. 不會有「HTML 字串裡插入未轉義動態資料」的 XSS 路徑(一律用 `textContent`,不用 `innerHTML` 拼字串)。

### 放置位置
`.gua-header`(卦頭區塊)內、`.question-block` 之後,用 `appendChild` 插入新元素 `.goldseal-block`。不占用/不修改現有 `.topbar .seal`(裝飾用「命」字圓章,語意不同,新元件一律用 `goldseal` 前綴命名)。

### 新增函式(全部用 DOM API 建構,不使用 `innerHTML` 拼接動態資料,杜絕 XSS)
```js
function mountGoldSeal(rec) {
  const header = document.querySelector(".gua-header");
  if (!header) return; // 防禦性:找不到掛載點時靜默跳過,不拋錯中斷詳情頁渲染

  const block = document.createElement("div");
  block.className = "goldseal-block";

  const isSealed = rec.golden_seal === true; // 嚴格布林比對,契約:Worker 保證 boolean(worker.js:61-62 /history、126-127 /log)
  if (isSealed) {
    renderSealedBadge(block, rec.golden_seal_time);
  } else {
    const btn = document.createElement("button");
    btn.type = "button";
    btn.className = "goldseal-btn";
    btn.id = "goldsealBtn";
    btn.textContent = "蓋金印"; // 定案文案,禁自創
    btn.addEventListener("click", () => {
      const logId = typeof rec.log_id === "string" && rec.log_id ? rec.log_id : null;
      if (!logId) {
        showGoldSealError(block);
        return;
      }
      sealGoldenSeal(logId, btn, block);
    });
    block.appendChild(btn);
  }

  header.appendChild(block);
}

function renderSealedBadge(block, sealedAtIso) {
  // 一律用 textContent,不用 innerHTML,杜絕 XSS(無論 fmtDate() 回傳什麼都不會被當標籤解析)
  block.className = "goldseal-block goldseal-sealed";
  block.textContent = `金印已落 ${fmtDate(sealedAtIso)}`; // 定案文案,禁自創
}

function showGoldSealError(block) {
  let toast = block.querySelector(".goldseal-toast");
  if (!toast) {
    toast = document.createElement("div");
    toast.className = "goldseal-toast";
    block.appendChild(toast);
  }
  toast.textContent = "金印未落,請稍後再試"; // 定案文案,禁自創,不分錯誤類型
}
```

### 互動邏輯
```js
async function sealGoldenSeal(logId, btn, block) {
  btn.disabled = true; // 送出即 disabled,防連打
  btn.textContent = "蓋章中……";
  const oldToast = block.querySelector(".goldseal-toast");
  if (oldToast) oldToast.remove(); // 重試前清掉舊錯誤 toast

  try {
    const res = await fetch(`${RELAY_URL}log/seal`, {
      method: "POST",
      headers: {
        "X-Line-AccessToken": liff.getAccessToken(),
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ log_id: logId }),
    });
    const data = await res.json().catch(() => null);

    // 不只信任 HTTP 200,還要驗證 body 的資料契約本身合法,否則視同失敗,不佯裝成功
    const sealedOk = res.ok && data && data.sealed === true &&
      (data.sealed_at === null || typeof data.sealed_at === "string");
    if (!sealedOk) {
      throw new Error("seal-failed"); // 內部用,畫面文案統一在 catch 內處理,不分錯誤類型
    }

    renderSealedBadge(block, data.sealed_at); // 樂觀更新只在通過驗證後才套用
  } catch (e) {
    console.warn("[goldseal] seal failed:", e); // 內部除錯用,不顯示給 TA
    btn.disabled = false;
    btn.textContent = "蓋金印";
    if (block && block.isConnected) {
      showGoldSealError(block); // 統一文案「金印未落,請稍後再試」
    }
  }
}
```

- 送出即 disable:`btn.disabled = true` 在 `fetch` 呼叫前立即執行,防連打;disabled 期間瀏覽器不會再觸發該按鈕的 `click` 事件,不需額外 debounce。
