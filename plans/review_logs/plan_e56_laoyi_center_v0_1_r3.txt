### 0.2 relay 既有七路由模式(workers/mingge-relay/worker.js)
七路由:`GET /history`、`GET /log`、`GET /study`、`POST /log/seal`、`POST /trace`、`POST /trigger/deepdive`、`POST /trigger/fupan

1. `Content-Type: application/json` 檢查
2. `request.json()` parse,失敗回 400
3. `X-Line-AccessToken` header 必帶,缺回 401
4. `resolveUserId(accessToken, LINE_CHANNEL_ID)`(第 654-677 行):先 `GET https://api.line.me/oauth2/v2.1/verify?access_toke
n=...` 驗 token 有效且 `client_id===LINE_CHANNEL_ID`,再 `GET https://api.line.me/v2/profile`(帶 `Authorization: Bearer <token
>`)取 `userId`/`displayName`。驗證失敗回 401。
5. payload 欄位白名單檢查(多餘欄位直接 400)
6. 業務邏輯 → 轉發下游(webhook 或未來的 Dify)
7. 統一用 `json(obj, status)` 輔助函式回應,CORS 由 `corsHeaders()` 處理

沒有任何既有路由呼叫過 Dify API(`grep -i dify` 零命中)── `/laoyi/chat` 是本 repo第一個 Dify 直連路由,呼叫慣例需自訂,採 Dify 官方 `chat-messages` blockin
g 端點慣例(`POST https://api.dify.ai/v1/chat-messages`,`Authorization: Bearer <app_key>`,body `{inputs:{}, query, response_
mode:"blocking", user, conversation_id}`,回應含 `answer`/`conversation_id`)。

### 1.5 對話送出(呼叫 C2 新路由)【r2 依 Codex r1 #4/#5/#16/#22/#23 修正】
```js

function laoyiSetComposerBusy(busy){

  laoyiInput.disabled=busy;
  laoyiSendBtn.disabled=busy;

async function laoyiSend(text){
  if(laoyiSending) return; // r1 #4
  var trimmed=(text||'').trim();
  if(!trimmed) return;
  if(trimmed.length>LAOYI_QUERY_MAX){
    laoyiAppendBubble('這一句太長了,請分段問(上限 '+LAOYI_QUERY_MAX+' 字)。','bot',{system:true});
    return; // r1 #16:前端先擋,不把超長內容當成功送出再等 400
  }
  laoyiSetComposerBusy(true);
  laoyiAppendBubble(trimmed,'user'); // textContent 寫入,見 laoyiAppendBubble 實作(r1 #22)
  laoyiInput.value='';
  var typing=laoyiAppendBubble('……','bot',{pending:true});
  var convIdAtSend=laoyiConvId; // r1 #4:鎖住送出當下的 conv id,避免競態下被別的回覆覆蓋
  var ok=false, answerText='';
  try{
    if(typeof liff==='undefined'){ throw new Error('LIFF unavailable'); }
    await ensureLiffInit();
    var accessToken = liff.getAccessToken ? liff.getAccessToken() : null;
    if(!accessToken){ throw new Error('no access token'); }
    var res = await fetch(RELAY_URL+'laoyi/chat',{
      method:'POST',
      headers:{'Content-Type':'application/json','X-Line-AccessToken':accessToken},
      body: JSON.stringify({query:trimmed, conversation_id: convIdAtSend||''})
    });
    if(!res.ok){ throw new Error('HTTP '+res.status); }
    var data = await res.json();
    if(!data || typeof data.answer!=='string' || !data.answer){ throw new Error('bad payload'); }
    laoyiConvId = data.conversation_id || laoyiConvId;
    answerText = data.answer;
    ok=true;
  }catch(e){
    answerText='老易這頭暫時聽不清，請再說一次，或稍後再試。';
    console.warn('[laoyi chat]',e && e.message?e.message:e);
  }
  laoyiReplaceBubble(typing, answerText); // textContent,不用 innerHTML(r1 #22)
  laoyiSetComposerBusy(false);
  if(ok){
    laoyiTurnCount++; // r1 #5:只有成功的一問一答才算一輪,token失效/503/網路失敗不計
    if(LAOYI_DECISION_RE.test(trimmed)){ laoyiAppendDecisionCta(); } // 承 1.7
    if(laoyiTurnCount>=25 && !laoyiClosingShown){ laoyiShowClosing(); laoyiClosingShown=true; } // r1 #5:idempotent,只出現
一次
  }
}

function laoyiComposerKeydown(e){ // r1 #23:Enter 送出/Shift+Enter 換行/IME 選字中不誤送
  if(e.key!=='Enter' || e.shiftKey) return;
  if(e.isComposing || e.keyCode===229) return;

}
```

應改為 guarded 操作，並建議將 `__laoyiInited=true` 移到必要 DOM 檢查通過後；否則過早初始化會永久阻止重試。
